## November 27, 2026 — #3946. Maximum Number of Items From Sale I

### 0/1 first purchases + cheapest extra copies

```javascript
function maximumSaleItems(items, budget) {
  const dp = Array(budget + 1).fill(0);
  let cheapestPrice = Infinity;

  for (const [factor, price] of items) {
    cheapestPrice = Math.min(cheapestPrice, price);

    // Count this purchased copy plus one free copy of every other divisible type.
    let firstPurchaseGain = 0;
    for (const [otherFactor] of items) {
      if (otherFactor % factor === 0) {
        firstPurchaseGain++;
      }
    }

    // Descending capacities allow this type's first-purchase bonus only once.
    for (let capacity = budget; capacity >= price; capacity--) {
      const skip = dp[capacity];
      const take = dp[capacity - price] + firstPurchaseGain;
      dp[capacity] = Math.max(skip, take);
    }
  }

  let answer = 0;
  for (let reserved = 0; reserved <= budget; reserved++) {
    const extras = Math.floor((budget - reserved) / cheapestPrice);
    answer = Math.max(answer, dp[reserved] + extras);
  }
  return answer;
}
```

```python
class Solution:
    def maximumSaleItems(self, items, budget):
        dp = [0] * (budget + 1)
        cheapest_price = float("inf")

        for factor, price in items:
            cheapest_price = min(cheapest_price, price)

            # Count this purchased copy plus one free copy of every other divisible type.
            first_purchase_gain = 0
            for other_factor, _ in items:
                if other_factor % factor == 0:
                    first_purchase_gain += 1

            # Descending capacities allow this type's first-purchase bonus only once.
            for capacity in range(budget, price - 1, -1):
                skip = dp[capacity]
                take = dp[capacity - price] + first_purchase_gain
                dp[capacity] = max(skip, take)

        answer = 0
        for reserved in range(budget + 1):
            extras = (budget - reserved) // cheapest_price
            answer = max(answer, dp[reserved] + extras)
        return answer
```
