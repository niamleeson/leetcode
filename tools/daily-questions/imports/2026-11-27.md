### DFS without memoization

```javascript
function maximumSaleItemsDFS(items, budget) {
    const rewards = Array(items.length).fill(0);

    // Calculate free rewards for each item
    for (let i = 0; i < items.length; i++) {
        const [factor] = items[i];

        for (let j = 0; j < items.length; j++) {
            const [otherFactor] = items[j];

            if (i !== j && otherFactor % factor === 0) {
                rewards[i]++;
            }
        }
    }

    function dfs(i, remaining, isFirstTimeBuy) {
        if (i === items.length || remaining === 0) {
            return 0;
        }

        const price = items[i][1];

        // Skip and move to the next item
        const skip = dfs(i + 1, remaining, true);

        if (remaining < price) {
            return skip;
        }

        // First purchase includes free rewards
        let value = 1;

        if (isFirstTimeBuy) {
            value += rewards[i];
        }

        // Stay at the current item to allow repeated purchases
        const take = value + dfs(i, remaining - price, false);

        return Math.max(skip, take);
    }

    return dfs(0, budget, true);
}
```

```python
class Solution:
    def maximumSaleItems(self, items, budget):
        import sys

        # A path can skip every type and buy budget // min_price copies.
        min_price = min(price for _, price in items)
        sys.setrecursionlimit(max(sys.getrecursionlimit(), len(items) + budget // min_price + 1000))
        rewards = [0] * len(items)

        # Calculate free rewards for each item.
        for i in range(len(items)):
            factor = items[i][0]

            for j in range(len(items)):
                other_factor = items[j][0]

                if i != j and other_factor % factor == 0:
                    rewards[i] += 1

        def dfs(i, remaining, is_first_time_buy):
            if i == len(items) or remaining == 0:
                return 0

            price = items[i][1]

            # Skip and move to the next item.
            skip = dfs(i + 1, remaining, True)

            if remaining < price:
                return skip

            # First purchase includes free rewards.
            value = 1

            if is_first_time_buy:
                value += rewards[i]

            # Stay at the current item to allow repeated purchases.
            take = value + dfs(i, remaining - price, False)

            return max(skip, take)

        return dfs(0, budget, True)
```

### Bottom-up DP (Mixed Knapsack)

```javascript
function maximumSaleItemsDP(items, budget) {
    const dp = Array(budget + 1).fill(0);

    for (let i = 0; i < items.length; i++) {
        const [factor, price] = items[i];

        // Calculate free rewards
        let reward = 0;

        for (let j = 0; j < items.length; j++) {
            const [otherFactor] = items[j];

            if (i !== j && otherFactor % factor === 0) {
                reward++;
            }
        }

        // 0/1 Knapsack: first purchase with rewards
        const firstPurchaseValue = 1 + reward;

        for (let c = budget; c >= price; c--) {
            dp[c] = Math.max(
                dp[c],
                dp[c - price] + firstPurchaseValue
            );
        }

        // Unbounded Knapsack: additional purchases
        for (let c = price; c <= budget; c++) {
            dp[c] = Math.max(
                dp[c],
                dp[c - price] + 1
            );
        }
    }

    return dp[budget];
}
```

```python
                    dp[c - price] + 1
```

### DFS with memoization

```javascript
function maximumSaleItemsMemo(items, budget) {
    const rewards = Array(items.length).fill(0);
    const memo = new Map();

    // Calculate free rewards for each item
    for (let i = 0; i < items.length; i++) {
        const [factor] = items[i];

        for (let j = 0; j < items.length; j++) {
            const [otherFactor] = items[j];

            if (i !== j && otherFactor % factor === 0) {
                rewards[i]++;
            }
        }
    }

    function dfs(i, remaining, isFirstTimeBuy) {
        if (i === items.length || remaining === 0) {
            return 0;
        }

        const key = `${i},${remaining},${isFirstTimeBuy}`;

        if (memo.has(key)) {
            return memo.get(key);
        }

        const price = items[i][1];

        // Skip and move to the next item
        const skip = dfs(i + 1, remaining, true);

        let take = 0;

        if (remaining >= price) {
            let value = 1;

            if (isFirstTimeBuy) {
                value += rewards[i];
            }

            take = value + dfs(i, remaining - price, false);
        }

        const result = Math.max(skip, take);

        memo.set(key, result);

        return result;
    }

    return dfs(0, budget, true);
}
```
