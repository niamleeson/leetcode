## November 27, 2026 — #3946. Maximum Number of Items From Sale I

### 0/1 first purchases + cheapest extra copies

```javascript
function maximumSaleItems(items, budget) {
  const dp = Array(budget + 1).fill(0);
  let minPrice = Infinity;

  for (const [factor, price] of items) {
    minPrice = Math.min(minPrice, price);

    // Count this purchased copy plus one free copy of every other divisible type.
    let gain = 0;
    for (const [other] of items) {
      if (other % factor === 0) {
        gain++;
      }
    }

    // Descending capacities allow this type's first-purchase bonus only once.
    for (let c = budget; c >= price; c--) {
      const skip = dp[c];
      const take = dp[c - price] + gain;
      dp[c] = Math.max(skip, take);
    }
  }

  let ans = 0;
  for (let cap = 0; cap <= budget; cap++) {
    const extras = Math.floor((budget - cap) / minPrice);
    ans = Math.max(ans, dp[cap] + extras);
  }
  return ans;
}
```

```python
class Solution:
    def maximumSaleItems(self, items, budget):
        dp = [0] * (budget + 1)
        min_price = float("inf")

        for factor, price in items:
            min_price = min(min_price, price)

            # Count this purchased copy plus one free copy of every other divisible type.
            gain = 0
            for other, _ in items:
                if other % factor == 0:
                    gain += 1

            # Descending capacities allow this type's first-purchase bonus only once.
            for c in range(budget, price - 1, -1):
                skip = dp[c]
                take = dp[c - price] + gain
                dp[c] = max(skip, take)

        ans = 0
        for cap in range(budget + 1):
            extras = (budget - cap) // min_price
            ans = max(ans, dp[cap] + extras)
        return ans
```
